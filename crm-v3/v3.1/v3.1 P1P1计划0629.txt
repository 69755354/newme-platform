### Step 1: 创建 `useSupabaseQuery` Hook

**目的**：统一数据加载逻辑，支持自动重试、loading/error状态管理。

#### 新建文件：`src/lib/supabaseQuery.ts`

```ts
// src/lib/supabaseQuery.ts
import { useEffect, useState } from 'react';
import type { PostgrestError } from '@supabase/supabase-js';

interface UseSupabaseQueryResult<T> {
  data: T | null;
  loading: boolean;
  error: string | null;
  refetch: () => void;
}

export function useSupabaseQuery<T>(
  queryFn: () => Promise<{ data: T | null; error: PostgrestError | null }>,
  deps: React.DependencyList = []
): UseSupabaseQueryResult<T> {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchData = async () => {
    setLoading(true);
    setError(null);
    try {
      const result = await queryFn();
      if (result.error) {
        console.error('[useSupabaseQuery] Query failed:', result.error);
        setError('加载失败，请重试');
        setData(null);
      } else {
        setData(result.data);
      }
    } catch (err) {
      console.error('[useSupabaseQuery] Unexpected error:', err);
      setError('网络错误，请检查连接');
      setData(null);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchData();
  }, deps);

  return { data, loading, error, refetch: fetchData };
}
```

### Step 4: 创建共享拖拽 Hook

#### 新建文件：`src/shared/hooks/usePipelineDragDrop.ts`

```ts
// src/shared/hooks/usePipelineDragDrop.ts
import { useState, useCallback } from 'react';
import { createClient } from '@/lib/supabase';
import { toast } from 'sonner';

interface DragDropHandlers {
  onDragStart: (e: React.DragEvent, leadId: string) => void;
  onDragOver: (e: React.DragEvent, stageKey: string) => void;
  onDragLeave: (e: React.DragEvent, stageKey: string) => void;
  onDrop: (e: React.DragEvent, targetStage: string) => Promise<void>;
  draggingLeadId: string | null;
  draggingOverStage: string | null;
}

export function usePipelineDragDrop(
  leads: any[],
  setLeads: (leads: any[]) => void,
  currentUserId: string | null
): DragDropHandlers {
  const [draggingLeadId, setDraggingLeadId] = useState<string | null>(null);
  const [draggingOverStage, setDraggingOverStage] = useState<string | null>(null);

  const onDragStart = useCallback((e: React.DragEvent, leadId: string) => {
    e.dataTransfer.setData('leadId', leadId);
    e.dataTransfer.effectAllowed = 'move';
    setDraggingLeadId(leadId);
  }, []);

  const onDragOver = useCallback((e: React.DragEvent, stageKey: string) => {
    e.preventDefault();
    setDraggingOverStage(stageKey);
  }, []);

  const onDragLeave = useCallback((e: React.DragEvent, stageKey: string) => {
    // 防止快速进出导致状态错乱
    setTimeout(() => {
      if (!e.currentTarget.contains(e.relatedTarget as Node)) {
        setDraggingOverStage(null);
      }
    }, 10);
  }, []);

  const onDrop = useCallback(async (e: React.DragEvent, targetStage: string) => {
    e.preventDefault();
    const leadId = e.dataTransfer.getData('leadId');
    setDraggingLeadId(null);
    setDraggingOverStage(null);

    if (!leadId || !currentUserId) return;

    const lead = leads.find(l => l.id === leadId);
    if (!lead || lead.stage === targetStage) return;

    const supabase = createClient();
    setUpdating(true);
    try {
      const { error } = await supabase
        .from('leads')
        .update({
          stage: targetStage,
          updated_at: new Date().toISOString(),
        })
        .eq('id', leadId);

      if (error) throw error;

      // Optimistic update
      setLeads(leads.map(l => 
        l.id === leadId ? { ...l, stage: targetStage } : l
      ));

      toast.success(`已移至 ${targetStage}`);
    } catch (err) {
      console.error('Stage update failed:', err);
      toast.error('移动失败，请重试');
    } finally {
      setUpdating(false);
    }
  }, [leads, setLeads, currentUserId]);

  return {
    onDragStart,
    onDragOver,
    onDragLeave,
    onDrop,
    draggingLeadId,
    draggingOverStage,
  };
}
```

```typescript
"use client";

import { useState, useEffect, useCallback, useRef } from "react";
import { createClient } from "@/lib/supabase";
import type { PostgrestError } from "@supabase/supabase-js";

// ─── Types ───
interface QueryState<T> {
  data: T | null;
  loading: boolean;
  error: string | null;
  status: "idle" | "loading" | "success" | "error";
}

interface QueryOptions {
  enabled?: boolean;
  retry?: number;
  retryDelay?: number;
  onSuccess?: (data: any) => void;
  onError?: (error: PostgrestError) => void;
}

// ─── useSupabaseQuery Hook ───
export function useSupabaseQuery<T = any>(
  queryFn: () => Promise<{ data: T | null; error: PostgrestError | null }>,
  deps: any[] = [],
  options: QueryOptions = {}
) {
  const {
    enabled = true,
    retry = 2,
    retryDelay = 1000,
    onSuccess,
    onError,
  } = options;

  const [state, setState] = useState<QueryState<T>>({
    data: null,
    loading: true,
    error: null,
    status: "idle",
  });

  const retryCount = useRef(0);
  const mountedRef = useRef(true);

  const execute = useCallback(async () => {
    if (!enabled) {
      setState({ data: null, loading: false, error: null, status: "idle" });
      return;
    }

    setState(prev => ({ ...prev, loading: true, error: null, status: "loading" }));

    try {
      const { data, error } = await queryFn();

      if (!mountedRef.current) return;

      if (error) {
        // 自动重试
        if (retryCount.current < retry) {
          retryCount.current += 1;
          setTimeout(execute, retryDelay * retryCount.current);
          return;
        }

        setState({
          data: null,
          loading: false,
          error: error.message,
          status: "error",
        });
        onError?.(error);
        return;
      }

      retryCount.current = 0;
      setState({
        data,
        loading: false,
        error: null,
        status: "success",
      });
      onSuccess?.(data);
    } catch (err) {
      if (!mountedRef.current) return;

      setState({
        data: null,
        loading: false,
        error: err instanceof Error ? err.message : "Unknown error",
        status: "error",
      });
    }
  }, [enabled, retry, retryDelay, ...deps]);

  useEffect(() => {
    mountedRef.current = true;
    execute();
    return () => {
      mountedRef.current = false;
    };
  }, [execute]);

  const refetch = useCallback(() => {
    retryCount.current = 0;
    execute();
  }, [execute]);

  return {
    ...state,
    refetch,
    isIdle: state.status === "idle",
    isLoading: state.status === "loading",
    isSuccess: state.status === "success",
    isError: state.status === "error",
  };
}

// ─── useSupabaseMutation Hook ───
export function useSupabaseMutation<T = any>(
  mutationFn: () => Promise<{ data: T | null; error: PostgrestError | null }>,
  options: {
    onSuccess?: (data: T) => void;
    onError?: (error: PostgrestError) => void;
  } = {}
) {
  const [state, setState] = useState<{
    loading: boolean;
    error: string | null;
  }>({ loading: false, error: null });

  const execute = useCallback(async () => {
    setState({ loading: true, error: null });

    try {
      const { data, error } = await mutationFn();

      if (error) {
        setState({ loading: false, error: error.message });
        options.onError?.(error);
        return { data: null, error };
      }

      setState({ loading: false, error: null });
      options.onSuccess?.(data as T);
      return { data, error: null };
    } catch (err) {
      const errorMsg = err instanceof Error ? err.message : "Unknown error";
      setState({ loading: false, error: errorMsg });
      return { data: null, error: errorMsg };
    }
  }, []);

  return {
    execute,
    ...state,
    isLoading: state.loading,
  };
}

// ─── useSupabaseSubscription Hook ───
export function useSupabaseSubscription<T = any>(
  channel: string,
  table: string,
  filter?: { column: string; value: string },
  callback?: (payload: T) => void
) {
  const supabase = createClient();

  useEffect(() => {
    const subscription = supabase
      .channel(channel)
      .on(
        "postgres_changes",
        {
          event: "*",
          schema: "public",
          table,
          ...(filter ? { filter: `${filter.column}=eq.${filter.value}` } : {}),
        },
        (payload) => {
          callback?.(payload as unknown as T);
        }
      )
      .subscribe();

    return () => {
      supabase.removeChannel(subscription);
    };
  }, [channel, table, filter?.column, filter?.value]);
}

// ─── useSupabasePagination Hook ───
export function useSupabasePagination<T = any>(
  queryFn: (page: number, pageSize: number) => Promise<{
    data: T[] | null;
    count: number | null;
    error: PostgrestError | null;
  }>,
  deps: any[] = [],
  options: { pageSize?: number; enabled?: boolean } = {}
) {
  const { pageSize = 20, enabled = true } = options;
  const [page, setPage] = useState(0);
  const [totalCount, setTotalCount] = useState(0);
  const [data, setData] = useState<T[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const totalPages = Math.ceil(totalCount / pageSize);

  const fetchPage = useCallback(async (pageNum: number) => {
    if (!enabled) return;
    setLoading(true);
    setError(null);

    try {
      const { data: result, count, error: err } = await queryFn(pageNum, pageSize);

      if (err) {
        setError(err.message);
        return;
      }

      setData(result || []);
      if (count !== null) setTotalCount(count);
    } catch (err) {
      setError(err instanceof Error ? err.message : "Unknown error");
    } finally {
      setLoading(false);
    }
  }, [enabled, pageSize, ...deps]);

  useEffect(() => {
    fetchPage(page);
  }, [page, fetchPage]);

  const nextPage = useCallback(() => {
    if (page < totalPages - 1) setPage(p => p + 1);
  }, [page, totalPages]);

  const prevPage = useCallback(() => {
    if (page > 0) setPage(p => p - 1);
  }, [page]);

  const goToPage = useCallback((pageNum: number) => {
    if (pageNum >= 0 && pageNum < totalPages) setPage(pageNum);
  }, [totalPages]);

  return {
    data,
    loading,
    error,
    page,
    totalPages,
    totalCount,
    pageSize,
    nextPage,
    prevPage,
    goToPage,
    refetch: () => fetchPage(page),
    hasNext: page < totalPages - 1,
    hasPrev: page > 0,
  };
}
```

```typescript
      hasError: false,
      error: null,
      errorInfo: null,

  }
```

```typescript
      setState({
        draggingId: null,
        draggingOver: null,
        isDragging: false,
        dragStartStage: null,
        dragStartIndex: null,
      });
    },
    []
  );

  // ─── 拖拽经过 ───
  const handleDragOver = useCallback(
    (e: React.DragEvent, stage: string) => {
      if (!enabled || !state.isDragging) return;
      e.preventDefault();
      e.dataTransfer.dropEffect = "move";

      // 使用 counter 防止频繁触发
      const key = stage;
      dragCounter.current[key] = (dragCounter.current[key] || 0) + 1;

      if (dragCounter.current[key] === 1) {
        setState(prev => ({ ...prev, draggingOver: stage }));
      }
    },
    [enabled, state.isDragging]
  );

  // ─── 拖拽离开 ───
  const handleDragLeave = useCallback(
    (e: React.DragEvent, stage: string) => {
      if (!enabled) return;

      const key = stage;
      dragCounter.current[key] = (dragCounter.current[key] || 0) - 1;

      if (dragCounter.current[key] <= 0) {
        dragCounter.current[key] = 0;
        setState(prev => ({
          ...prev,
          draggingOver: prev.draggingOver === stage ? null : prev.draggingOver,
        }));
      }
    },
    [enabled]
  );

  // ─── 放置 ───
  const handleDrop = useCallback(
    async (e: React.DragEvent, toStage: string, toIndex: number) => {
      if (!enabled || !state.draggingId) return;
      e.preventDefault();

      const leadId = state.draggingId;
      const fromStage = state.dragStartStage;
      const fromIndex = state.dragStartIndex;

      if (!fromStage || fromStage === toStage) {
        // 同一阶段内排序
        if (fromIndex !== null && fromIndex !== toIndex) {
          const result: DropResult = {
            leadId,
            fromStage,
            toStage,
            fromIndex,
            toIndex,
          };

          // 防抖处理
          lastUpdateRef.current = result;
          if (debounceTimer.current) {
            clearTimeout(debounceTimer.current);
          }

          debounceTimer.current = setTimeout(async () => {
            if (lastUpdateRef.current) {
              await onReorder?.(
                lastUpdateRef.current.leadId,
                lastUpdateRef.current.fromIndex,
                lastUpdateRef.current.toIndex
              );
              lastUpdateRef.current = null;
            }
          }, debounceMs);
        }
        return;
      }

      // 跨阶段移动
      setUpdating(true);
      try {
        const result: DropResult = {
          leadId,
          fromStage,
          toStage,
          fromIndex: fromIndex || 0,
          toIndex,
        };

        await onStageChange?.(result);

        // 更新数据库
        const { error } = await supabase
          .from("leads")
          .update({
            stage: toStage,
            updated_at: new Date().toISOString(),
            stage_order: toIndex,
            last_drag_at: new Date().toISOString(),
          })
          .eq("id", leadId);

        if (error) {
          console.error("[usePipelineDragDrop] Update failed:", error);
          toast.error("Failed to move lead");
          return;
        }

        toast.success(`Moved to ${toStage}`);
      } catch (err) {
        console.error("[usePipelineDragDrop] Drop error:", err);
        toast.error("Failed to move lead");
      } finally {
        setUpdating(false);
        setState(prev => ({
          ...prev,
          draggingId: null,
          draggingOver: null,
          isDragging: false,
        }));
      }
    },
    [enabled, state, debounceMs, onStageChange, onReorder, supabase]
  );

  // ─── 获取阶段内的 leads ───
  const getLeadsByStage = useCallback(
    (stage: string) => {
      return leads
        .filter(lead => lead.stage === stage)
        .sort((a: any, b: any) => {
          // 按 stage_order 排序
          if (a.stage_order !== undefined && b.stage_order !== undefined) {
            return a.stage_order - b.stage_order;
          }
          // 降级到 created_at
          return new Date(b.created_at).getTime() - new Date(a.created_at).getTime();
        });
    },
    [leads]
  );

  // ─── 获取阶段统计 ───
  const getStageStats = useCallback(
    (stage: string) => {
      const stageLeads = getLeadsByStage(stage);
      const totalValue = stageLeads.reduce(
        (sum: number, lead: any) => sum + (lead.quotation_value || 0),
        0
      );
      return {
        count: stageLeads.length,
        totalValue,
        leads: stageLeads,
      };
    },
    [getLeadsByStage]
  );

  return {
    // State
    draggingId: state.draggingId,
    draggingOver: state.draggingOver,
    isDragging: state.isDragging,
    updating,

    // Handlers
    handleDragStart,
    handleDragEnd,
    handleDragOver,
    handleDragLeave,
    handleDrop,

    // Utilities
    getLeadsByStage,
    getStageStats,

    // Reset
    reset: useCallback(() => {
      setState({
        draggingId: null,
        draggingOver: null,
        isDragging: false,
        dragStartStage: null,
        dragStartIndex: null,
      });
      dragCounter.current = {};
    }, []),
  };
}
```

```typescript
import { NextRequest, NextResponse } from "next/server";

export async function GET(request: NextRequest) {
  const searchParams = request.nextUrl.searchParams;
  const leadId = searchParams.get("lead_id");

  const offset = parseInt(searchParams.get("offset") || "0", 10);

  if (!leadId) {
    return NextResponse.json(
      { error: "lead_id is required" },
      { status: 400 }
    );
  }

  const supabase = createClient();

  try {
    // 并行查询多个数据源
    const [followUpLogs, tasks, events, chatMessages] = await Promise.all([
      supabase
        .from("follow_up_logs")
        .select("id, contact_type, summary, user_id, created_at")
        .eq("lead_id", leadId)
        .order("created_at", { ascending: false })
        .limit(limit)
        .offset(offset),
      
      supabase
        .from("tasks")
        .select("id, title, due_at, status, created_at")
        .eq("lead_id", leadId)
        .order("created_at", { ascending: false })

        .offset(offset),
      
      supabase

        .eq("lead_id", leadId)

        .offset(offset),
      
      supabase
        .from("chat_messages")
        .select("id, content, direction, created_at")
        .eq("lead_id", leadId)
        .order("created_at", { ascending: false })
        .limit(limit)
        .offset(offset),

        createdAt: log.created_at,

        id: task.id,

        title: task.title,
        description: `Status: ${task.status}`,
        createdAt: task.created_at,
```

```typescript
// src/shared/hooks/usePipelineDragDrop.ts
import { useState, useCallback, useRef } from 'react';
import { toast } from 'sonner';

interface DragItem {
  id: string;
  type: 'lead';
}

interface DropResult {
  stage: string;
}

export function usePipelineDragDrop() {
  const [draggingOver, setDraggingOver] = useState<string | null>(null);
  const dragCounter = useRef<Record<string, number>>({});

  const handleDragStart = useCallback((e: React.DragEvent, item: DragItem) => {
    e.dataTransfer.setData('text/plain', JSON.stringify(item));
    e.dataTransfer.effectAllowed = 'move';
    (e.currentTarget as HTMLElement).classList.add('opacity-40');
  }, []);

  const handleDragEnd = useCallback((e: React.DragEvent) => {
    (e.currentTarget as HTMLElement).classList.remove('opacity-40');
    setDraggingOver(null);
    Object.keys(dragCounter.current).forEach(key => {
      dragCounter.current[key] = 0;
    });
  }, []);

  const handleDragOver = useCallback((e: React.DragEvent, stage: string) => {
    e.preventDefault();
    setDraggingOver(stage);
    dragCounter.current[stage] = (dragCounter.current[stage] || 0) + 1;
  }, []);

  const handleDragLeave = useCallback((e: React.DragEvent, stage: string) => {
    dragCounter.current[stage] = Math.max(0, (dragCounter.current[stage] || 0) - 1);
    if (dragCounter.current[stage] === 0) {
      setDraggingOver(null);
    }
  }, []);

  const handleDrop = useCallback(
    async (e: React.DragEvent, stage: string, updateLeadStage: (id: string, stage: string) => Promise<void>) => {
      e.preventDefault();
      const data = e.dataTransfer.getData('text/plain');
      const item: DragItem = JSON.parse(data);
      
      if (item.type === 'lead') {
        try {
          await updateLeadStage(item.id, stage);
          toast.success('Lead moved successfully');
        } catch (error) {
          toast.error('Failed to move lead');
        }
      }
      
      setDraggingOver(null);
      Object.keys(dragCounter.current).forEach(key => {
        dragCounter.current[key] = 0;
      });
    },
    []
  );

  return {
    draggingOver,
    handleDragStart,
    handleDragEnd,
    handleDragOver,
    handleDragLeave,
    handleDrop,
  };
}
```

```typescript
// src/components/DashboardErrorBoundary.tsx
import React from 'react';
import { AlertTriangle, RefreshCw } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { useLanguage } from '@/lib/i18n/LanguageContext';

interface DashboardErrorBoundaryProps {
  children: React.ReactNode;
}

interface ErrorState {
  hasError: boolean;
  error?: Error;
}

export class DashboardErrorBoundary extends React.Component<
  DashboardErrorBoundaryProps,
  ErrorState
> {
  constructor(props: DashboardErrorBoundaryProps) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): ErrorState {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('Dashboard Error:', error, errorInfo);
  }

  handleRetry = () => {
    this.setState({ hasError: false, error: undefined });
  };

  render() {
    if (this.state.hasError) {
      return (
        <div className="error-boundary-fallback">
          <AlertTriangle className="icon" />
          <h2>Something went wrong</h2>
          <p>We encountered an unexpected error. Please try again.</p>
          <Button onClick={this.handleRetry}>
            <RefreshCw className="w-4 h-4" />
            Try Again
          </Button>
        </div>
      );
    }

    return this.props.children;
  }
}
```

### 4. src/shared/hooks/usePipelineDragDrop.ts
```typescript
// src/shared/hooks/usePipelineDragDrop.ts
import { useState, useCallback, useRef } from 'react';
import { toast } from 'sonner';

interface DragItem {
  id: string;
  type: 'lead';
}

interface DropResult {
  stage: string;
}

export function usePipelineDragDrop() {
  const [draggingOver, setDraggingOver] = useState<string | null>(null);
  const dragCounter = useRef<Record<string, number>>({});

  const handleDragStart = useCallback((e: React.DragEvent, item: DragItem) => {
    e.dataTransfer.setData('text/plain', JSON.stringify(item));
    e.dataTransfer.effectAllowed = 'move';
    (e.currentTarget as HTMLElement).classList.add('opacity-40');
  }, []);

  const handleDragEnd = useCallback((e: React.DragEvent) => {
    (e.currentTarget as HTMLElement).classList.remove('opacity-40');
    setDraggingOver(null);
    Object.keys(dragCounter.current).forEach(key => {
      dragCounter.current[key] = 0;
    });
  }, []);

  const handleDragOver = useCallback((e: React.DragEvent, stage: string) => {
    e.preventDefault();
    setDraggingOver(stage);
    dragCounter.current[stage] = (dragCounter.current[stage] || 0) + 1;
  }, []);

  const handleDragLeave = useCallback((e: React.DragEvent, stage: string) => {
    dragCounter.current[stage] = Math.max(0, (dragCounter.current[stage] || 0) - 1);
    if (dragCounter.current[stage] === 0) {
      setDraggingOver(null);
    }
  }, []);

  const handleDrop = useCallback(
    async (e: React.DragEvent, stage: string, updateLeadStage: (id: string, stage: string) => Promise<void>) => {
      e.preventDefault();
      const data = e.dataTransfer.getData('text/plain');
      const item: DragItem = JSON.parse(data);
      
      if (item.type === 'lead') {
        try {
          await updateLeadStage(item.id, stage);
          toast.success('Lead moved successfully');
        } catch (error) {
          toast.error('Failed to move lead');
        }
      }
      
      setDraggingOver(null);
      Object.keys(dragCounter.current).forEach(key => {
        dragCounter.current[key] = 0;
      });
    },
    []
  );

  return {
    draggingOver,
    handleDragStart,
    handleDragEnd,
    handleDragOver,
    handleDragLeave,
    handleDrop,
  };
}
```

**结论:** `profiles` 表本身**没有** `email` 列。代码需要通过 `JOIN auth.users` 来获取用户的 email。

**依据:**
- Schema 速查表中 `profiles` 表的字段列表**不包含** `email` 列。
- 备注中明确说明 `email` 字段的来源是 `(from auth.users)`，这意味着它并非 `profiles` 表的物理列，而是指可以通过关联 `auth.users` 表获得。
- 这是 Supabase 的标准做法：用户认证信息（如 email）存储在 `auth.users` 表中，而用户业务信息（如 `full_name`, `role`）存储在 `public.profiles` 表中。

**代码实现建议：**
在 `src/app/(dashboard)/leads/page.tsx` 和 `src/app/(dashboard)/pipeline/page.tsx` 等需要用户 email 的地方，Supabase Query 应使用类似以下的 JOIN 写法：
```sql
-- 示例：获取 leads 及其 assigned_to 用户的 email
SELECT 
  leads.*,
  assigned_profile.full_name,
  auth_users.email
FROM leads
LEFT JOIN profiles AS assigned_profile ON leads.assigned_to = assigned_profile.id
LEFT JOIN auth.users AS auth_users ON leads.assigned_to = auth_users.id;
```
或者，如果 `profiles` 表有 `email` 的同步机制（例如通过触发器从 `auth.users` 同步），则可以直接查询 `profiles.email`。但根据提供的 schema，**没有**这样的同步机制，因此必须使用 `JOIN auth.users`。
